Comptes et mots de passe
Les comptes DG et agents se gèrent dans la console Firebase (Authentication). Chaque personne a son e-mail et son mot de passe.














  rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function profil() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    function connecte() {
      return request.auth != null
        && exists(/databases/$(database)/documents/users/$(request.auth.uid));
    }
    function dg() { return connecte() && profil().role == 'dg'; }

    // Profil de chaque personne : lecture de son propre profil seulement.
    // Les profils se créent à la main dans la console Firebase.
    match /users/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false;
    }

    // Visible par tous les comptes, modifiable par le DG seulement.
    match /catalog/{id}  { allow read: if connecte(); allow write: if dg(); }
    match /stockin/{id}  { allow read: if connecte(); allow write: if dg(); }

    // Prix d'achat et achats : DG seulement.
    match /costs/{id}     { allow read, write: if dg(); }
    match /purchases/{id} { allow read, write: if dg(); }

    // Ventes, dépenses, paiements : DG et agents.
    match /sales/{id}    { allow read, write: if connecte(); }
    match /expenses/{id} { allow read, write: if connecte(); }
    match /payments/{id} { allow read, write: if connecte(); }
  }
}
