Configuration et mise en ligne

Cette page est conçue pour être publiée sur GitHub Pages. Les identifiants Firebase publics (firebaseConfig) doivent être ceux de votre propre projet. La sécurité réelle est appliquée par les règles Firestore, pas par les boutons cachés de cette page.

Règles Firestore à installer dans Firebase

Avant de publier et d'utiliser avec de vrais stocks, faites vérifier ces règles selon votre schéma. Elles séparent les prix d'achat et les bénéfices des données visibles par Marleine.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function myUser() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    function active() { return signedIn() && myUser().active == true; }
    function isDG() { return active() && myUser().role == 'DG'; }
    function isAgent() { return active() && myUser().role == 'AGENT'; }

    match /users/{uid} {
      allow read: if signedIn() && request.auth.uid == uid || isDG();
      allow write: if false; // attribuer les rôles manuellement depuis Firebase Console
    }
    match /catalog/{id} {
      allow read: if active();
      allow create, delete: if isDG();
      // L'agent peut uniquement diminuer le stock; il ne peut pas modifier le prix de vente.
      allow update: if isDG() || (isAgent()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['stock','updatedAt'])
        && request.resource.data.stock <= resource.data.stock);
    }
    match /costs/{id} {
      allow read, write: if isDG(); // prix d'achat réservés au Responsable
    }
    match /sales/{id} {
      allow read: if isDG() || (isAgent() && resource.data.sellerUid == request.auth.uid);
      allow create: if active()
        && request.resource.data.sellerUid == request.auth.uid
        && request.resource.data.keys().hasOnly(['productId','productName','code','qty','unitPrice','total','client','sellerUid','sellerName','createdAt'])
        && request.resource.data.qty is int && request.resource.data.qty > 0
        && request.resource.data.unitPrice is number
        && request.resource.data.total == request.resource.data.qty * request.resource.data.unitPrice;
      allow update, delete: if isDG();
    }
    match /stockin/{id} { allow read, write: if isDG(); }
    match /expenses/{id} { allow read, write: if isDG(); }
    match /{document=**} { allow read, write: if false; }
  }
}
Important : les règles ci-dessus constituent une base de départ. Pour une gestion de stock avec plusieurs vendeurs, une Cloud Function est recommandée pour valider atomiquement chaque vente et la diminution du stock. Ne mettez jamais les prix d'achat dans les documents catalog lisibles par Marleine.
